\documentclass[11pt,oneside]{article}

\usepackage[paperwidth=6in,paperheight=9in,
            top=0.95in,bottom=1.05in,inner=0.9in,outer=0.9in]{geometry}
\usepackage{fontspec}
\usepackage{xcolor}
\usepackage{microtype}
\usepackage{ragged2e}
\usepackage{needspace}
\usepackage{fancyhdr}
\usepackage{titlesec}
\usepackage[hidelinks,pdfusetitle]{hyperref}
\usepackage{etoolbox}

\defaultfontfeatures{Ligatures=TeX, Scale=MatchLowercase}

% Generated by build.py: \newfontfamily\<mood>Quote, \<mood>Body,
% \definecolor{<mood>}{HTML}{...}, plus \setmainfont.
\input{fonts.tex}

\definecolor{ink}{HTML}{2b2622}
\definecolor{soft}{HTML}{7a7268}
\color{ink}

% ── Running heads ────────────────────────────────────────────────────────────
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyfoot[C]{\color{soft}\footnotesize\thepage}
\fancyhead[L]{\color{soft}\footnotesize\scshape\MakeLowercase{\leftmark}}
\renewcommand{\sectionmark}[1]{}

% ── Ornaments ────────────────────────────────────────────────────────────────
\newcommand{\ornament}[1]{% colour name
  \par\vspace{1.6em}
  \begin{center}\color{#1}\rule{2.2em}{0.4pt}\end{center}
  \vspace{1.6em}}

% ── The entry environment ────────────────────────────────────────────────────
% \begin{entry}{mood}{quote font cmd}{body font cmd}{scale}{style}
%   \quotetext{...}
%   \attribution{where}{date}
%   \commentary{...}
% \end{entry}
\newlength{\quotemeasure}
\setlength{\quotemeasure}{0.86\textwidth}
\newlength{\bodymeasure}
\setlength{\bodymeasure}{0.78\textwidth}

\newenvironment{entry}[5]{%
  \needspace{9\baselineskip}%
  \def\entrymood{#1}%
  \def\entryquotefont{#2}%
  \def\entrybodyfont{#3}%
  \def\entryscale{#4}%
  \def\entrystyle{#5}%
  \par\noindent
}{%
  \ornament{\entrymood}%
}

\newcommand{\quotetext}[1]{%
  \begin{center}
    \begin{minipage}{\quotemeasure}
      \Centering
      \entryquotefont
      \ifdefstring{\entrystyle}{italic}{\itshape}{}%
      \fontsize{\dimexpr\entryscale\dimexpr 11pt\relax\relax}{\dimexpr\entryscale\dimexpr 14.5pt\relax\relax}\selectfont
      #1\par
    \end{minipage}
  \end{center}
}

\newcommand{\attribution}[2]{% where, date  (the book is the document, so no author here)
  \ifboolexpr{test{\ifblank{#1}} and test{\ifblank{#2}}}{}{%
    \par\vspace{0.9em}
    \begin{center}
      \entrybodyfont\color{\entrymood}\small
      \ifblank{#1}{}{\textemdash\ #1}%
      \ifblank{#2}{}{\ifblank{#1}{}{\\[0.2em]}{\color{soft}\scriptsize\scshape\addfontfeatures{LetterSpace=8}#2}}%
    \end{center}}%
}

\newcommand{\commentary}[1]{%
  \par\vspace{1.3em}
  \begin{center}
    \begin{minipage}{\bodymeasure}
      \entrybodyfont\small
      \setlength{\parindent}{1.2em}
      \setlength{\parskip}{0pt}
      \noindent #1\par
    \end{minipage}
  \end{center}
}

% ── Title page (one per book) ───────────────────────────────────────────────
% \booktitlepage{mood}{quote font cmd}{title}{author}{translator}{year}{read}{series}
\newcommand{\booktitlepage}[8]{%
  \begin{titlepage}
    \centering
    \vspace*{0.26\textheight}
    {\color{soft}\small\scshape\addfontfeatures{LetterSpace=16}\MakeLowercase{#4}\par}
    \vspace{1.6em}
    {#2\color{ink}\fontsize{30pt}{36pt}\selectfont\itshape #3\par}
    \vspace{1.4em}
    {\color{#1}\rule{2.2em}{0.4pt}\par}
    \vspace{1.2em}
    {\color{soft}\small
      \ifblank{#5}{}{translated by #5\par\vspace{0.3em}}%
      \ifblank{#6}{}{#6\par}}
    \vfill
    {\color{soft}\footnotesize\scshape\addfontfeatures{LetterSpace=10}\MakeLowercase{#8}%
      \ifblank{#7}{}{\ \textperiodcentered\ read #7}\par}
    \vspace*{0.08\textheight}
  \end{titlepage}
  \setcounter{page}{2}%
}

% \booknotes{body font cmd}{text} — optional paragraph(s) about the book as a whole
\newcommand{\booknotes}[2]{%
  \begin{center}
    \begin{minipage}{\bodymeasure}
      #1\small\setlength{\parindent}{1.2em}\noindent #2\par
    \end{minipage}
  \end{center}
  \ornament{soft}%
}
